SELECT
  hd1.Country AS `Country`,
  hd1.Site AS `Site`,
  hd1.Subject AS `Subject`,
  hd1.FormName AS `Form Name`,
  hd1.Seq AS `Seq`,
  hd1.LocationType AS `Location Type`,
  hd1.Location AS `Location`,
  hd1.LocationSiteDetails AS `Location site details`
FROM
(
  SELECT
    `@HDR`.`Site`.`Country` AS Country,
    `@HDR`.`Site`.`Name` AS Site,
    `@HDR`.`Subject`.`Name` AS Subject,
    `@Form`.`Name` AS FormName,
    CDDHFACATENDIT.GRPID AS Seq,
    CDDHFACATENDIT.LOCTYPE AS LocationType,
    CDDHFACATENDIT.LOC AS Location,
    CDDHFACATENDIT.LOCSITEDETAILS AS LocationSiteDetails,
    `@HDR`.`Subject`.`ID` AS SubjID
  FROM `EDC`.`CDDHFM` CDDHFACATENDIT
  WHERE `@Form`.`Status` = 'submitted__v'
) hd1
WHERE EXISTS (
  SELECT 1
  FROM
  (
    SELECT
      CDDHFACATENDIT2.GRPID AS Seq2,
      CDDHFACATENDIT2.LOCTYPE AS LocationType2,
      CDDHFACATENDIT2.LOC AS Location2,
      `@HDR`.`Subject`.`ID` AS SubjID2
    FROM `EDC`.`CDDHFM` CDDHFACATENDIT2
    WHERE `@Form`.`Status` = 'submitted__v'
  ) hd2
  WHERE hd2.SubjID2 = hd1.SubjID
    AND hd2.LocationType2 = hd1.LocationType
    AND hd2.Location2 = hd1.Location
    AND hd2.Seq2 <> hd1.Seq
)

UNION ALL

SELECT
  cd1.Country AS `Country`,
  cd1.Site AS `Site`,
  cd1.Subject AS `Subject`,
  cd1.FormName AS `Form Name`,
  cd1.Seq AS `Seq`,
  cd1.LocationType AS `Location Type`,
  cd1.Location AS `Location`,
  cd1.LocationSiteDetails AS `Location site details`
FROM
(
  SELECT
    `@HDR`.`Site`.`Country` AS Country,
    `@HDR`.`Site`.`Name` AS Site,
    `@HDR`.`Subject`.`Name` AS Subject,
    `@Form`.`Name` AS FormName,
    ODBHFCFCATIT.GRPID AS Seq,
    ODBHFCFCATIT.LOCTYPE AS LocationType,
    ODBHFCFCATIT.LOC AS Location,
    ODBHFCFCATIT.LOCSITEDETAILS AS LocationSiteDetails,
    `@HDR`.`Subject`.`ID` AS SubjID
  FROM `EDC`.`ODBHFFM` ODBHFCFCATIT
  WHERE `@Form`.`Status` = 'submitted__v'
) cd1
WHERE EXISTS (
  SELECT 1
  FROM
  (
    SELECT
      ODBHFCFCATIT2.GRPID AS Seq2,
      ODBHFCFCATIT2.LOCTYPE AS LocationType2,
      ODBHFCFCATIT2.LOC AS Location2,
      `@HDR`.`Subject`.`ID` AS SubjID2
    FROM `EDC`.`ODBHFFM` ODBHFCFCATIT2
    WHERE `@Form`.`Status` = 'submitted__v'
  ) cd2
  WHERE cd2.SubjID2 = cd1.SubjID
    AND cd2.LocationType2 = cd1.LocationType
    AND cd2.Location2 = cd1.Location
    AND cd2.Seq2 <> cd1.Seq
)
ORDER BY `Country`, `Site`, `Subject`, `Seq`
