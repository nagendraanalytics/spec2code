SELECT
  base.Country AS `Country`,
  base.SiteMnemonic AS `Site`,
  base.SubjectNumber AS `Subject Number`,
  base.Cohort AS `Cohort`,
  base.ExpectedType AS `Expected Tissue Acquisition Type`,
  base.ExpectedVisit AS `Expected Visit`
FROM
(
  SELECT
    `@HDR`.`Site`.`Country` AS Country,
    `@HDR`.`Site`.`Number` AS SiteMnemonic,
    `@HDR`.`Subject`.`Name` AS SubjectNumber,
    `@HDR`.`Subject`.`ID` AS SubjectID,
    CriteriaFm.CriteriaValue AS Cohort,
    CriteriaFm.ExpectedType AS ExpectedType,
    CriteriaFm.ExpectedVisit AS ExpectedVisit
  FROM
  (

/* Placeholder derived table representing the supporting-document driven expected Visit / Tissue Acquisition Type / Criteria combinations. In production this would be generated per row of ONC_TSIF_001_supp_<Vxxxxxxx>.csv (Criteria Form, Criteria Item, Criteria Value, Visit, Tissue Acquisition Type). Since no metadata-confirmed Criteria Form/Item is available for this study, SR.Cohort is used illustratively and must be replaced/expanded per the actual supporting doc content at runtime. */

    SELECT
      `@HDR`.`Subject`.`ID` AS SubjectID2,
      SASCORRESSNUMIT AS CriteriaValue,
      'Archival' AS ExpectedType,
      'Screening' AS ExpectedVisit
    FROM `EDC`.`SAFM`
    WHERE `@Form`.`Status` = 'submitted__v'
  ) CriteriaFm
  WHERE `@HDR`.`Subject`.`ID` = CriteriaFm.SubjectID2
    AND NOT EXISTS (
      SELECT 1
      FROM `EDC`.`TSIFFM` t
      WHERE `@HDR`.`Subject`.`ID` = `@HDR`.`Subject`.`ID`
        AND t.`@Form`.`Status` = 'submitted__v'
        AND (t.DELETEDFM = 'N' OR t.DELETEDFM IS NULL)
        AND t.TSIFCETYPBPSYIT = CriteriaFm.ExpectedType
        AND t.TSIFCESTDTCSSSDIT <> 'Not Obtained'
        AND `@HDR`.`Event`.`Name` = CriteriaFm.ExpectedVisit
    )
    AND NOT EXISTS (
      SELECT 1
      FROM `EDC`.`DOVFM` d
      WHERE `@HDR`.`Subject`.`ID` = `@HDR`.`Subject`.`ID`
        AND d.`@Form`.`Status` = 'submitted__v'
        AND (d.DELETEDFM = 'N' OR d.DELETEDFM IS NULL)
        AND d.DOVSERRORVIT = 'Y'
        AND `@HDR`.`Event`.`Name` = CriteriaFm.ExpectedVisit
    )
) base
ORDER BY
  base.Country,
  base.SiteMnemonic,
  base.SubjectNumber,
  base.Cohort
