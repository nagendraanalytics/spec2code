SELECT
  vendor.Country AS `Country`,
  vendor.SiteNumber AS `Site Number`,
  vendor.ScreeningNumber AS `Screening number`,
  vendor.SubjectNumber AS `Subject Number`,
  vendor.VendorName AS `Vendor`,
  vendor.VendorSiteID AS `Site ID or number`,
  vendor.VendorScreeningID AS `Screening ID or Number`,
  vendor.VendorSubjectID AS `Subject ID or Number`
FROM
(
  SELECT
    `@HDR`.`Site`.`Country` AS Country,
    `@HDR`.`Site`.`Number` AS SiteNumber,
    SASCORRESSNUMIT AS ScreeningNumber,
    `@HDR`.`Subject`.`Name` AS SubjectNumber,
    `@Form`.`Name` AS VendorName,
    ENRMSUBJNUMIT AS VendorSiteID,
    SASCORRESSNUMIT AS VendorScreeningID,
    SADMSUBJIDIT AS VendorSubjectID,
    `@HDR`.`Subject`.`ID` AS SubjectRefID
  FROM `EDC`.`SAFM`
  WHERE `@Form`.`Status` = 'submitted__v'
    AND (DELETEDFM = 'N' OR DELETEDFM IS NULL)
    AND SASCORRESSNUMIT IS NOT NULL
    AND ENRMSUBJNUMIT IS NOT NULL
) vendor
WHERE EXISTS (
  SELECT 1
  FROM `EDC`.`GEFCF` g
  WHERE g.`@HDR`.`Subject`.`ID` = vendor.SubjectRefID
    AND g.`@Form`.`Status` = 'submitted__v'
    AND (g.DELETEDFM = 'N' OR g.DELETEDFM IS NULL)
    AND g.GEFCFSCRNIT = vendor.ScreeningNumber
    AND (g.GEFCFSITEIT <> vendor.VendorSiteID OR g.GEFCFSITEIT IS NULL)
)
ORDER BY
  vendor.Country,
  vendor.SiteNumber,
  vendor.ScreeningNumber,
  vendor.SubjectNumber,
  vendor.VendorName,
  vendor.VendorSiteID,
  vendor.VendorScreeningID,
  vendor.VendorSubjectID
