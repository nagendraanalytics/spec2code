SELECT
  base.`Country` AS `Country`,
  base.`Site` AS `Site`,
  base.`Screening Number` AS `Screening Number`,
  base.`Subject Number` AS `Subject Number`,
  base.`Vendor` AS `Vendor`,
  base.`Kit ID` AS `Kit ID`,
  base.`Slide Cut Date` AS `Slide Cut Date`,
  base.`Slide Stain Date` AS `Slide Stain Date`
FROM
(
  SELECT
    `@HDR`.`Site`.`Country` AS `Country`,
    `@HDR`.`Site`.`Number` AS `Site`,
    SA.`Screening Number` AS `Screening Number`,
    SA.`Subject Number` AS `Subject Number`,
    cut.VendorName AS `Vendor`,
    cut.KitID AS `Kit ID`,
    DATE_FORMAT(RAWDATE(cut.ResultValue), '%d-%m-%Y') AS `Slide Cut Date`,
    DATE_FORMAT(RAWDATE(stain.ResultValue), '%d-%m-%Y') AS `Slide Stain Date`,
    `@HDR`.`Subject`.`ID` AS SubjectID
  FROM
  (
    SELECT
      `@HDR`.`Subject`.`ID` AS SubjectID,
      GEFCFKITID AS KitID,
      GEFCFVENDORNAME AS VendorName,
      CASE WHEN GEFCFBLINDFL = 'B' THEN 'xxxx' ELSE GEFCFORRES END AS ResultValue
    FROM `EDC`.`GEFCFFM`
    WHERE `@Form`.`Status` = 'submitted__v'
      AND (DELETEDFM = 'N' OR DELETEDFM IS NULL)
      AND GEFCFTESTSHORTNAME = 'SLDCUTDT'
  ) cut
  INNER JOIN
  (
    SELECT
      `@HDR`.`Subject`.`ID` AS SubjectID,
      GEFCFKITID AS KitID,
      GEFCFVENDORNAME AS VendorName,
      CASE WHEN GEFCFBLINDFL = 'B' THEN 'xxxx' ELSE GEFCFORRES END AS ResultValue
    FROM `EDC`.`GEFCFFM`
    WHERE `@Form`.`Status` = 'submitted__v'
      AND (DELETEDFM = 'N' OR DELETEDFM IS NULL)
      AND GEFCFTESTSHORTNAME = 'SLSTNDT'
  ) stain
  ON cut.SubjectID = stain.SubjectID
    AND cut.KitID = stain.KitID
  INNER JOIN
  (
    SELECT
      `@HDR`.`Subject`.`ID` AS SubjectID,
      SASCORRESSNUMIT AS `Screening Number`,
      SASUBJIDIT AS `Subject Number`
    FROM `EDC`.`SAFM`
    WHERE `@Form`.`Status` = 'submitted__v'
  ) SA
  ON SA.SubjectID = cut.SubjectID
  WHERE RAWDATE(stain.ResultValue) > SUBDATE(RAWDATE(cut.ResultValue), INTERVAL -@WindowOfDays DAY)
) base
ORDER BY
  base.`Country`,
  base.`Site`,
  base.`Subject Number`
