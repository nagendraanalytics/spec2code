/*************************************************************************************************;
%*************************************************************************************************;
%* Report ID   : ABC011_JR                                                                         ;
%* Report Title: Subjects With More Than One AE With Action of Drug Withdrawn                     ;
%* Date Created: ******                                                                       ;
%* Programmer  : ******                                                                       ;
%*************************************************************************************************;
%* Report Description:                                                                            ;
%* This report identifies subjects with more than one AE with action of Drug Withdrawn            ;
%*************************************************************************************************;
%* Report Logic:                                                                                  ;
%* This report will output subjects who have more than one records of "Drug Withdrawn" for Action ;
%* Taken With Study Medication on the AE Form.                                                    ;
%*                                                                                                ;
%* Notes:                                                                                         ;
%* 1. SAS Names for Action Taken With Study Medication = ACN1 and/or ACN2 and/or ACN3 and/or ACN4 ;
%* and/or ACN5 and/or ACN6 and/or ACN7 and/or ACN8 and/or ACN9 and/or ACN10 and/or AEACNBLD       ;
%* 2.Report should not display deleted records.                                                   ;
%* 3.If "Randomized only" parameter selection checkbox is checked, the report should only output  ;
%* subjects with Randomization number that is not null.                                           ;
%*
%*************************************************************************************************;
%* Program Flow:                                                                                  ;
%* 1. Get Data                                                                                    ;
%* 2. Apply Logic                                                                                 ;
%* 3. Generate output to Excel                                                                    ;
%*************************************************************************************************;
%* Macros Used: getdata3                                                                          ;
%*                                                                                                                ;
%*************************************************************************************************;
%* Revision History:                                                                              ;
%* Version 1 - Initial Version - Chuyi Han                                                        ;

%*************************************************************************************************;*/



%macro ABC011_JR ;

%let repid=%upcase(ABC011_JR );
%let vernum=6;
%let repdesc=This report identifies subjects with more than one AE with action of Drug Withdrawn.;

%dvrheader;

/*************************coding started ***********************************************/
/* Code updated to fetch the data from IRV_Studyverion_form*/
data _null_;
        call symput('mdval','BA_'!!left(trim(upcase("&studyid")))!!'_INF_MD_'!!left(trim(&lifecyc)));
run;

%let mdval1 = %bquote('&mdval');

proc sql noprint;
    connect to oracle as prddb (user = &iruser pw = &irpaswd path = &irconnct);
    exec (begin mrkintgsys.pkg_merck_util.set_context(&mdval1); end;) by prddb;

    create table irv_studyversion_forms as
    select * from connection to prddb
    (
    	select formmnemonic from &mdval..irv_studyversion_forms
        where upper(trim(formmnemonic))='VAE'
    );
	disconnect from prddb;
quit;

proc sql noprint;
select count(formmnemonic) into :cd from irv_studyversion_forms;
quit;

/* If VAE form doesn't exists, pick AE form */
%if &cd=0 %then %do;
/*===========================================================AE FORM====================================================================*/
        %* ------Get AE form data using getdata3 macro --- *;
        %getData3(usnm =&iruser, pwd = &irpaswd,pth =&irconnct,stnm =&stdynm,env = &lifecyc,rmarea = '_RM2_',
        tbl=rm_ae|*,
        antbl=,
        bntbl=);

	/* Version 5 : check if blinded columns SMB and AEACNBLD exists */

	%* --- check if smb is exists in the table --- *;
	%if %varexist(rm_ae,smb)>0 %then %do;
			

			%* --- If more than 1 "DRUG WITHDRAWN" within each medication for blided study medication--- *;
				proc sql noprint;
                    create table bld as
                    select distinct sitecountry,sitemnemonic,mercknum,aespid,aeterm,aeacnbld as acn,smb as sm
                    from rm_ae
                    where upcase(strip(aeacnbld)) = "DRUG WITHDRAWN"
					and smb is not null;

	
                    create table final_aeacnbld as
                    select sitecountry,sitemnemonic,mercknum,aespid,aeterm,acn,sm
                    from bld
                    group by mercknum,sm
                    having count(aeterm)>1;
                quit;	
	%end;
	%* --- extract the data from acn&i and sm&i--- *;


			%let i = 1;
                                %do i = 1 %to 10;
                                        proc sql noprint;
                                            create table final&i as
                                            select distinct sitecountry,sitemnemonic,mercknum,aespid,aeterm,acn&i as acn,sm&i as sm
                                            from rm_ae
                                            where upcase(strip(acn&i)) = "DRUG WITHDRAWN"
											and sm&i is not null
											/*Version 6 : Jagatha start*/
											%if %varexist(rm_ae,sm1h)>0 %then %do;
											union

											select distinct sitecountry,sitemnemonic,mercknum,aespid,aeterm,acn&i as acn,sm&i.h as sm
                                            from rm_ae
                                            where upcase(strip(acn&i)) = "DRUG WITHDRAWN"
											and sm&i.h is not null
											%end;
											;
											/*Version 6 : Jagatha end*/
                                        quit;

                                        proc append data = final&i base = final_all;
                                        run;

										proc datasets nolist;
										delete final&i;
										run;
                                %end;

	

	data final_ae;
	set final:;
	run;

	proc sort data = final_ae nodupkey;
    by sitecountry sitemnemonic mercknum aespid sm;
	run;

	proc sql noprint;
	create table final_all as
     select sitecountry,sitemnemonic,mercknum,aespid,aeterm,acn,sm
     from final_ae
     group by mercknum,sm
     having count(aeterm)>1;
   quit;

%end;

/*==========================================================VAE FORM====================================================================*/

/* If VAE form  exists */
%if &cd > 0 %then %do;

        %* --- 1. Get VAE form data using getdata3 macro --- *;;
        %getData3(usnm =&iruser, pwd = &irpaswd,pth =&irconnct,stnm =&stdynm,env = &lifecyc,rmarea = '_RM2_',
        tbl=rm_vae|*,
        antbl=,
        bntbl=);

	/* Version 5 : check if blinded columns SMB and AEACNBLD exists */
	%* --- check if smb is exists in the table --- *;
	%if %varexist(rm_vae,smb)>0 %then %do;

			%* --- If more than 1 "DRUG WITHDRAWN" within each medication for blided study medication--- *;
				proc sql noprint;
                  create table bld as
                  select distinct sitecountry,sitemnemonic,mercknum,aespid,aeterm,aeacnbld as acn,smb as sm
                  from rm_vae
                  where upcase(strip(aeacnbld)) = "DRUG WITHDRAWN"
				  and smb is not null;


				  create table final_aeacnbld as
                  select sitecountry,sitemnemonic,mercknum,aespid,aeterm,acn,sm
                  from bld
                  group by mercknum,sm
                  having count(aeterm)>1;
                quit;
		
	%end;

	%* ----extract the data from acn&i and sm&i --- *;

			%let i = 1;
                                %do i = 1 %to 5;
                                        proc sql noprint;
                                           create table final&i as
                                           select distinct sitecountry,sitemnemonic,mercknum,aespid,aeterm,acn&i as acn,sm&i as sm
                                           from rm_vae
                                           where upcase(strip(acn&i)) = "DRUG WITHDRAWN"
										   and sm&i is not null
										   /*Version 6 : Jagatha start*/
										   %if %varexist(rm_vae,sm1h)>0 %then %do;
											union

											select distinct sitecountry,sitemnemonic,mercknum,aespid,aeterm,acn&i as acn,sm&i.h as sm
                                            from rm_vae
                                            where upcase(strip(acn&i)) = "DRUG WITHDRAWN"
											and sm&i.h is not null
											%end;
											;
											/*Version 6 : Jagatha end*/
                                        quit;

                                        proc append data = final&i base = final_all;
                                        run;
										
										proc datasets nolist;
										delete final&i;
										run;

                                %end;
										
	

	data final_vae;
	set final:;
	run;

	proc sort data = final_vae nodupkey;
    by sitecountry sitemnemonic mercknum aespid sm;
	run;

	proc sql noprint;
	create table final_all as
     select sitecountry,sitemnemonic,mercknum,aespid,aeterm,acn,sm
     from final_vae
     group by mercknum,sm
     having count(aeterm)>1;
   quit;

%end;

%if &Randomized = Y %then %do;
        data final_all;
            set final_all;
            where index(mercknum,'--') > 0;
        run;
%end;

proc sort data = final_all;
    by sitecountry sitemnemonic mercknum aespid;
run;


%* --- 3. Generate output to Excel --- *;
%obscnt(final_all);

%if &nobs=0 %then %do;
    ods tagsets.excelxp
        %if &putfil=NON %then &putfilnm;
        options (embedded_titles       =  'yes'
                embedded_footnotes     =  'yes'
                print_footer           =  'Page &amp;P of &amp;N'
                sheet_name             =  'AE011'
                orientation            =  'landscape'
                absolute_column_width  =  '100');



    options formchar='|_' missing=' ' pageno=1 nobyline ls=180;



    title6  'AE011_JR - Subjects With More Than One AE With Action of Drug Withdrawn';
    title7  "Randomized Only : &Randomized";


    data nobs;
        message = "No records meet the test condition";
    run;


    proc report data=nobs headline headskip spacing=1 split='*' missing;
        column message;
        define message    / ' ';
    run;
%end; /* End of &nobs=0 */

%else %do;
    ods tagsets.excelxp
    %if &putfil=NON %then &putfilnm;
    options (embedded_titles       =  'yes'
            embedded_footnotes     =  'yes'
            print_footer           =  'Page &amp;P of &amp;N'
            sheet_name             =  'AE011_JR'
            orientation            =  'landscape'
            absolute_column_width  =  '15,10,15,10,20,25');

        options formchar='|_' missing=' ' pageno=1 nobyline ls=180;

        title6  'ABC011_JR  - Subjects With More Than One AE With Action of Drug Withdrawn';
        title7  "Randomized Only : &Randomized";

        proc report data=final_all headline headskip spacing=1 split='*' missing;
            column sitecountry sitemnemonic mercknum aespid aeterm sm;
            define sitecountry      / display   width=15 'Country'    flow;
            define sitemnemonic     / display   width=10 'Site' style(column)={tagattr='format:@'};
            define mercknum         / display    width=15 'Subject Number' style(column)={tagattr='format:@'};
            define aespid           / display   width=10 'AE Seq';
            define aeterm           / display   width=20 'AE Term' flow;
            define sm               / display   width=25 'Drug Withdrawn Study Medication' flow;
        run;



%end; /* End of &nobs > 0 */



%mend ABC011_JR ;
