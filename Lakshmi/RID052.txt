SELECT
  base.Country AS `Country`,
  base.Site AS `Site`,
  base.SubjectNumber AS `Subject Number`,
  base.VisitMnemonic AS `Visit Mnemonic`,
  base.IVRContactDate AS `IVR Contact Date`,
  base.DaysSinceIVRContact AS `Days since IVR Contact`,
  base.DispositionStatus AS `Disposition Status`
FROM
(
  SELECT
    `@HDR`.`Site`.`Country` AS Country,
    `@HDR`.`Site`.`Number` AS Site,
    `@HDR`.`Subject`.`Name` AS SubjectNumber,
    `@HDR`.`Event`.`Name` AS VisitMnemonic,
    DATE_FORMAT(SRDSDTCAASMTROIT, '%d-%m-%Y') AS IVRContactDate,
    RAWDATE(SRDSDTCAASMTROIT) AS IVRContactDateRaw,
    DATEDIFF(CURRENT_DATE, RAWDATE(SRDSDTCAASMTROIT)) AS DaysSinceIVRContact,
    DS01DSTERMCMPIT AS DispositionStatus,
    `@HDR`.`Subject`.`ID` AS SubjectID
  FROM `EDC`.`SRIVRSFM`
  WHERE `@Form`.`Status` = 'submitted__v'
    AND SRDSDTCAASMTROIT IS NOT NULL
    AND RAWDATE(SRDSDTCAASMTROIT) < SUBDATE(CURRENT_DATE, INTERVAL @DaysWindow DAY)
    AND (@RandomizedOnly = 0 OR `@HDR`.`Subject`.`Name` LIKE '%--%')
    AND NOT EXISTS (
      SELECT 1
      FROM `EDC`.`DOVFM` dov
      WHERE dov.`@HDR`.`Subject`.`ID` = `@HDR`.`Subject`.`ID`
        AND dov.`@Form`.`Status` = 'submitted__v'
        AND dov.`@HDR`.`Event`.`Name` = `@HDR`.`Event`.`Name`
        AND (dov.DOVSERRORVIT IS NULL OR dov.DOVSERRORVIT = 'N')
        AND dov.DOV IS NOT NULL
    )
) base
ORDER BY
  base.Country,
  base.Site,
  base.SubjectNumber,
  base.VisitMnemonic
