SELECT
  `@HDR`.`Site`.`Country` AS `Country`,
  `@HDR`.`Site`.`Name` AS `Site`,
  `@HDR`.`Subject`.`Name` AS `Subject Number`,
  `@Form`.`Name` AS `eCRF`
FROM `EDC`.`SAFM`
WHERE `@Form`.`Status` = 'submitted__v'
  AND `SAFM`.`SAELIGCRITIT` = 'YES'
  AND NOT EXISTS (
    SELECT 1
    FROM `EDC`.`DOVFM` dov
    WHERE dov.`@Form`.`Status` = 'submitted__v'
      AND dov.`@HDR`.`Subject`.`ID` = `SAFM`.`@HDR`.`Subject`.`ID`
      AND dov.`DOVSERRORVIT` = true
  )
  AND EXISTS (
    SELECT 1
    FROM `EDC`.`IEFM` ie
    WHERE ie.`@Form`.`Status` = 'submitted__v'
      AND ie.`@HDR`.`Subject`.`ID` = `SAFM`.`@HDR`.`Subject`.`ID`
      AND (
        ie.`IEDTC01IT` IS NOT NULL
        OR ie.`IEPROTVIT` IS NOT NULL
        OR ie.`IETESTCDEXIT` IS NOT NULL
        OR ie.`IETESTCDINIT` IS NOT NULL
        OR ie.`IETESTCDEX01CB` IS NOT NULL
        OR ie.`IETESTCDEX02CB` IS NOT NULL
        OR ie.`IETESTCDEX03CB` IS NOT NULL
        OR ie.`IETESTCDEX04CB` IS NOT NULL
        OR ie.`IETESTCDEX05CB` IS NOT NULL
        OR ie.`IETESTCDEX06CB` IS NOT NULL
        OR ie.`IETESTCDEX07CB` IS NOT NULL
        OR ie.`IETESTCDEX08CB` IS NOT NULL
        OR ie.`IETESTCDEX09CB` IS NOT NULL
        OR ie.`IETESTCDEX10CB` IS NOT NULL
        OR ie.`IETESTCDEX11CB` IS NOT NULL
        OR ie.`IETESTCDEX12CB` IS NOT NULL
        OR ie.`IETESTCDEX13CB` IS NOT NULL
        OR ie.`IETESTCDEX14CB` IS NOT NULL
        OR ie.`IETESTCDEX15CB` IS NOT NULL
        OR ie.`IETESTCDEX16CB` IS NOT NULL
        OR ie.`IETESTCDEX17CB` IS NOT NULL
        OR ie.`IETESTCDEX18CB` IS NOT NULL
        OR ie.`IETESTCDEX19CB` IS NOT NULL
        OR ie.`IETESTCDEX20CB` IS NOT NULL
        OR ie.`IETESTCDEX21CB` IS NOT NULL
        OR ie.`IETESTCDIN01CB` IS NOT NULL
        OR ie.`IETESTCDIN02CB` IS NOT NULL
        OR ie.`IETESTCDIN03CB` IS NOT NULL
        OR ie.`IETESTCDIN04CB` IS NOT NULL
        OR ie.`IETESTCDIN05CB` IS NOT NULL
        OR ie.`IETESTCDIN06CB` IS NOT NULL
        OR ie.`IETESTCDIN07CB` IS NOT NULL
        OR ie.`IETESTCDIN08CB` IS NOT NULL
        OR ie.`IETESTCDIN09CB` IS NOT NULL
      )
  )
ORDER BY
  `Country`,
  `Site`,
  `Subject Number`
